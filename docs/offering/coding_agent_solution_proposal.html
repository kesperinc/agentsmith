B. Qwen2.5-Coder-32B vLLM ServingRuntime YAML (`qwen-vllm-servingruntime.yaml`)

apiVersion: serving.kserve.io/v1alpha1
kind: ServingRuntime
metadata:
  name: qwen-coder-runtime
  namespace: rhoai-coding-agent
spec:
  supportedModelFormats:
    - name: vLLM
  containers:
    - name: kserve-container
      image: quay.io/modh/vllm:rhoai-2.14
      command: ["python3", "-m", "vllm.entrypoints.openai.api_server"]
      args:
        - "--model=Qwen/Qwen2.5-Coder-32B-Instruct"
        - "--tensor-parallel-size=2"
        - "--quantization=fp8"
        - "--max-model-len=32768"
        - "--enable-prefix-caching"
      resources:
        limits: { nvidia.com/gpu: "2", memory: "64Gi" }
        requests: { nvidia.com/gpu: "2", memory: "32Gi" }
